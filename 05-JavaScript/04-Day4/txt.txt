function countMultiples(start, end, divisor) {
  let count = 0;

  for (let number = start; number <= end; number++) {
    if (number % divisor !== 0) {
      continue;
    }

    count++;
  }

  return count;
}

console.log(countMultiples(1, 10, 3));
